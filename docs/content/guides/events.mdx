A mod is a set of handlers. It registers them as it loads, and the server
calls each one when its moment comes: a command typed in chat, a frame, a
player who joined, a hit about to land.

<Samples>

```ts
import { command, log, onFrame, onPlayerJoin, onStart, onWorld } from 'modlock'

onStart((args) => log(`started with ${args.join(' ')}`))
onWorld((map) => log(`world ${map} loaded`))
onPlayerJoin(({ connection }) => log(`${connection.name} joined`))
onFrame((frame) => {
  // Runs once per server frame, at frame.timeSeconds.
})
command('hello', (player, args) => {
  player.chat(`You said ${args}`)
})
```

```luau
local modlock = require("@modlock")

modlock.onStart(function(args)
	modlock.log(`started with {table.concat(args, " ")}`)
end)
modlock.onWorld(function(map)
	modlock.log(`world {map} loaded`)
end)
modlock.onPlayerJoin(function(joined)
	modlock.log(`{joined.connection.name} joined`)
end)
modlock.onFrame(function(frame)
	-- Runs once per server frame, at frame.timeSeconds.
end)
modlock.command("hello", function(player, args)
	player:chat(`You said {args}`)
end)
```

```python
import modlock


@modlock.on_start
def started(args: list[str]) -> None:
    modlock.log(f"started with {' '.join(args)}")


@modlock.on_world
def loaded(map_name: str) -> None:
    modlock.log(f"world {map_name} loaded")


@modlock.on_player_join
def joined(event: modlock.PlayerJoinedEvent) -> None:
    modlock.log(f"{event.connection.name} joined")


@modlock.on_frame
def frame(frame: modlock.FrameEvent) -> None:
    pass  # Runs once per server frame, at frame.time_seconds.


@modlock.command("hello")
def hello(player: modlock.Player, args: str) -> None:
    player.chat(f"You said {args}")
```

```go
package main

import "github.com/paralin/modlock/mod"

func init() {
	mod.OnStart(func(args []string) { mod.Log("started with", args) })
	mod.OnWorld(func(mapName string) { mod.Log("world", mapName, "loaded") })
	mod.OnPlayerJoin(func(joined *mod.PlayerJoinedEvent) {
		mod.Log(joined.GetConnection().GetName(), "joined")
	})
	mod.OnFrame(func(frame *mod.FrameEvent) {
		// Runs once per server frame, at frame.GetTimeSeconds().
	})
	mod.Command("hello", func(p mod.Player, args string) {
		_ = p.Chat("You said " + args)
	})
}

func main() {}
```

</Samples>

## Every event

The table names each call as TypeScript and Luau spell it. Python uses
snake_case decorators, such as `@modlock.on_player_join`, and Go exported
functions, such as `mod.OnPlayerJoin`.

| Call                                                | Runs the handler                                              |
| --------------------------------------------------- | ------------------------------------------------------------- |
| `command(name, handler)`                            | On the frame after a player types `/name` in chat.            |
| `onChat(handler)`                                   | On the frame after each chat line, with `all` for all chat.   |
| `onStart(handler)`                                  | When the server starts the mod, with its `--arg` values.      |
| `onWorld(handler)`                                  | As each world loads, with its map; at start if one has.       |
| `onFrame(handler)`                                  | Once per server frame.                                        |
| `onPlayerJoin(handler)`                             | As each player the mod serves is ready; at start for each.    |
| `onPlayerLeave(handler)`                            | Once per joined player who leaves, while their hero reads.    |
| `onDamage(handler)`, `onDamaged(handler)`           | Before each hit lands, to block or change it; after it lands. |
| `onLanded(handler)`                                 | When a hero lands under the manifest's movement model.        |
| `onSettingChanged(handler)`                         | When a player changes a setting outside the mod.              |
| `watchProjectiles(options)`, `onLaunch`, `onImpact` | On a watched projectile's first frame, and on its impact.     |

## Joins and leaves

A join and a leave carry the player's `connection`, with their slot, SteamID64
and name. A leave also says whether the player `disconnected` from the server,
false when they stay but the mod stops serving them, and the `reason` the
server gave, such as `'kicked'` or `'disconnectByUser'`, when it gave one. Each
joined player leaves exactly once, and a player who disconnects leaves before
the server removes them, so their hero still reads:

<Samples>

```ts
import { log, onPlayerLeave } from 'modlock'

onPlayerLeave(({ connection, disconnected, reason }) => {
  if (disconnected) {
    log(`${connection.name} left (${reason ?? 'no reason'})`)
  }
})
```

```luau
modlock.onPlayerLeave(function(left)
	if left.disconnected then
		modlock.log(`{left.connection.name} left`)
	end
end)
```

```python
@modlock.on_player_leave
def left(event: modlock.PlayerLeftEvent) -> None:
    if event.disconnected:
        modlock.log(f"{event.connection.name} left")
```

```go
mod.OnPlayerLeave(func(left *mod.PlayerLeftEvent) {
	if left.GetDisconnected() {
		mod.Log(left.GetConnection().GetName(), "left:", left.GetReason())
	}
})
```
