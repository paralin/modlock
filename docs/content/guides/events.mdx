title: Events
description: Engine events and how to register handlers for them.
---

Your mod registers its handlers as it loads. The server calls each handler when
its moment comes, such as a chat command, a frame, a player joining or a hit
about to land.

<Samples>

```ts
import { command, log, onFrame, onPlayerJoin, onStart, onWorld } from 'modlock'

onStart((args) => log(`started with ${args.join(' ')}`))
onWorld((map) => log(`world ${map} loaded`))
onPlayerJoin(({ connection }) => log(`${connection.name} joined`))
onFrame((frame) => {
  // Runs once per server frame, at frame.timeSeconds.
})
command('hello', (player, args) => {
  player.chat(`You said ${args}`)
})
```

```luau
local modlock = require("@modlock")

modlock.onStart(function(args)
	modlock.log(`started with {table.concat(args, " ")}`)
end)
modlock.onWorld(function(map)
	modlock.log(`world {map} loaded`)
end)
modlock.onPlayerJoin(function(joined)
	modlock.log(`{joined.connection.name} joined`)
end)
modlock.onFrame(function(frame)
	-- Runs once per server frame, at frame.timeSeconds.
end)
modlock.command("hello", function(player, args)
	player:chat(`You said {args}`)
end)
```

```python
import modlock


@modlock.on_start
def started(args: list[str]) -> None:
    modlock.log(f"started with {' '.join(args)}")


@modlock.on_world
def loaded(map_name: str) -> None:
    modlock.log(f"world {map_name} loaded")


@modlock.on_player_join
def joined(event: modlock.PlayerJoinedEvent) -> None:
    modlock.log(f"{event.connection.name} joined")


@modlock.on_frame
def frame(frame: modlock.FrameEvent) -> None:
    pass  # Runs once per server frame, at frame.time_seconds.


@modlock.command("hello")
def hello(player: modlock.Player, args: str) -> None:
    player.chat(f"You said {args}")
```

```go
package main

import "github.com/paralin/modlock/mod"

func init() {
	mod.OnStart(func(args []string) { mod.Log("started with", args) })
	mod.OnWorld(func(mapName string) { mod.Log("world", mapName, "loaded") })
	mod.OnPlayerJoin(func(joined *mod.PlayerJoinedEvent) {
		mod.Log(joined.GetConnection().GetName(), "joined")
	})
	mod.OnFrame(func(frame *mod.FrameEvent) {
		// Runs once per server frame, at frame.GetTimeSeconds().
	})
	mod.Command("hello", func(p mod.Player, args string) {
		_ = p.Chat("You said " + args)
	})
}

func main() {}
```

</Samples>

## Every event

Python spells each event as a `snake_case` decorator, such as
`@modlock.on_player_join`. Go spells it as an exported function, such as
`mod.OnPlayerJoin`. In Python, an event that takes a name takes it in the
decorator, as `@modlock.on_modifier_event("PRE_UNIT_HEALED")` does.

| Call                                                | Runs the handler                                                                           |
| --------------------------------------------------- | ------------------------------------------------------------------------------------------ |
| `command(name, handler)`                            | On the frame after a player types `/name` in chat.                                         |
| `onChat(handler)`                                   | On the frame after each chat line, with `all` for all chat.                                |
| `onPing(handler)`                                   | Before each ping reaches anyone, to block it or mute its voice; see [Pings](#pings).       |
| `onStart(handler)`                                  | When the server starts the mod, with the words after `--` on the command line.             |
| `onWorld(handler)`                                  | As each world loads, with its map, and at start if one already loaded.                     |
| `onFrame(handler)`                                  | Once per server frame.                                                                     |
| `onPlayerJoin(handler)`                             | As each player finishes joining, and at start for each player already joined.              |
| `onSpawn(handler)`                                  | When a player's first hero enters play and on each respawn, with the hero.                 |
| `onPlayerAway(handler)`, `onPlayerReturn(handler)`  | When a joined player's connection drops while the game holds their seat; when they return. |
| `onPlayerLeave(handler)`                            | Once per joined player whose seat ends, while their hero can still be read.                |
| `onDamage(handler)`, `onDamaged(handler)`           | Before each hit lands, to block or change it; after it lands.                              |
| `onKill(handler)`                                   | After each hit that kills a hero or a unit; see [Damage](/guides/damage#react-to-a-kill).  |
| `onAbilityCast(handler)`                            | After each ability or item the game executes.                                              |
| `onHeal(handler)`                                   | After each heal; see [Heals, levels and melee](#heals-levels-and-melee).                   |
| `onLevelChange(handler)`                            | As each hero reaches a new level.                                                          |
| `onMelee(handler)`, `onParry(handler)`              | After each melee swing; after each parry that catches one.                                 |
| `onModifierEvent(name, handler)`                    | Each time the game fires the named modifier event.                                         |
| `onLanded(handler)`                                 | When a hero lands under the manifest's movement model.                                     |
| `onSettingChanged(handler)`                         | When a player changes a setting outside the mod.                                           |
| `watchProjectiles(options)`, `onLaunch`, `onImpact` | On a watched projectile's first frame, and on its impact.                                  |

## A player's stages

Most mods need only the join, to greet the player, and the spawn, to act on
the hero.

| Stage      | What is true                                                                                           | How the mod sees it                                                        |
| ---------- | ------------------------------------------------------------------------------------------------------ | -------------------------------------------------------------------------- |
| Connecting | The player holds a slot and is still loading the map. Nothing sent to them shows yet.                  | In the [player list](/guides/players#list-the-players) with `ready` false. |
| Joined     | The player finished loading. Chat, announcements and interfaces reach them. They may not have a hero yet. | `onPlayerJoin`, once. `ready` is true from then on.                        |
| In play    | The player's hero is alive in the world, so the mod can move it, equip it and read it.                 | `onSpawn`, again on each respawn.                                          |
| Away       | The connection dropped, and the game holds the seat, hero and team for the player's return.            | `onPlayerAway`, then `onPlayerReturn` if they come back. `away` is true.   |
| Left       | The seat ended, and the mod no longer serves the player.                                               | `onPlayerLeave`, once for each join.                                       |

Between joined and in play, the player is picking a team or a hero, or the game
has not placed the hero yet. [`player.seat`](/guides/toolkit#seat-the-players)
skips the team and hero screens. A call that needs a hero, such as a teleport,
fails before the hero spawns. So make that call in `onSpawn`, which fires for
the first hero and for each respawn. A bot joins as soon as it is added.

## Joins and leaves

When a player's connection drops, the player is away rather than gone; the game
holds their seat, hero, and team for five minutes. If they reconnect within that
window, they resume as the same `Player`. Modlock suppresses UI updates while a
player is away and restores them when they return. A mod that pauses while
someone is missing (such as during a ready check) can track disconnects with
`onPlayerAway` and `onPlayerReturn`.

Each joined player leaves exactly once, when their seat ends. The seat ends
when they leave the game, when the mod stops serving them, when a stranger
takes the held slot, or when the five minutes run out. Your leave handler can
still read their hero. A moment later, Modlock kills the hero, keeps it from
respawning and takes the controller off its team. So a mod never has to clean
up a hero that a player left behind in the world:

<Samples>

```ts
import { log, onPlayerLeave } from 'modlock'

onPlayerLeave(({ connection, disconnected, reason }) => {
  if (disconnected) {
    log(`${connection.name} left (${reason ?? 'no reason'})`)
  }
})
```

```luau
modlock.onPlayerLeave(function(left)
	if left.disconnected then
		modlock.log(`{left.connection.name} left`)
	end
end)
```

```python
@modlock.on_player_leave
def left(event: modlock.PlayerLeftEvent) -> None:
    if event.disconnected:
        modlock.log(f"{event.connection.name} left")
```

```go
mod.OnPlayerLeave(func(left *mod.PlayerLeftEvent) {
	if left.GetDisconnected() {
		mod.Log(left.GetConnection().GetName(), "left:", left.GetReason())
	}
})
```

</Samples>

| Field          | On             | Holds                                                                                    |
| -------------- | -------------- | ---------------------------------------------------------------------------------------- |
| `connection`   | join and leave | The slot, SteamID64 and name.                                                            |
| `disconnected` | leave          | Whether the player left the server; false when they stay but the mod stops serving them. |
| `reason`       | leave          | The server's reason, such as `'kicked'` or `'disconnectByUser'`, when it gave one.       |

## Spawns

`onSpawn` passes your handler the player and the hero now in play:

<Samples>

```ts
import { log, onSpawn } from 'modlock'

onSpawn((player, pawn) => {
  log(`slot ${player.slot} spawned with ${pawn.health} health`)
})
```

```luau
modlock.onSpawn(function(player, pawn)
	modlock.log(`slot {player.slot} spawned with {pawn.health} health`)
end)
```

```python
@modlock.on_spawn
def spawned(player: modlock.Player, pawn: modlock.Pawn) -> None:
    modlock.log(f"slot {player.slot} spawned with {pawn.health} health")
```

```go
mod.OnSpawn(func(player mod.Player, pawn *mod.Pawn) {
	mod.Log("slot", player.Slot(), "spawned with", pawn.GetHealth(), "health")
})
```

</Samples>

The library detects spawns from the engine's `PLAYER_RESPAWNED` modifier event
(once on initial spawn and again on each respawn) and runs your handler before
that frame's `onFrame` handlers. Team changes do not fire `onSpawn`. A server
that cannot watch modifier events does not report spawns.

## Casts and modifier events

Mods can subscribe to ability casts and any engine modifier event by name:

<Samples>

```ts
import { log, onAbilityCast, onModifierEvent } from 'modlock'

onAbilityCast((cast) => {
  if (cast.slot >= 0) {
    log(`slot ${cast.player.slot} cast ${cast.name}`)
  }
})

onModifierEvent('PRE_UNIT_HEALED', (event) => {
  log(`entity ${event.caster} is about to heal entity ${event.target}`)
})
```

```luau
modlock.onAbilityCast(function(cast)
	if cast.slot >= 0 then
		modlock.log(`slot {cast.player.slot} cast {cast.name}`)
	end
end)

modlock.onModifierEvent("PRE_UNIT_HEALED", function(event)
	modlock.log(`entity {event.caster} is about to heal entity {event.target}`)
end)
```

```python
@modlock.on_ability_cast
def cast(event: modlock.AbilityCastEvent) -> None:
    if event.slot >= 0:
        modlock.log(f"slot {event.player.slot} cast {event.name}")


@modlock.on_modifier_event("PRE_UNIT_HEALED")
def healing(event: modlock.ModifierFiredEvent) -> None:
    modlock.log(f"entity {event.caster} is about to heal entity {event.target}")
```

```go
mod.OnAbilityCast(func(cast *mod.AbilityCastEvent) {
	if cast.GetSlot() >= 0 {
		mod.Log("slot", cast.GetPlayer(), "cast", cast.GetName())
	}
})

mod.OnModifierEvent("PRE_UNIT_HEALED", func(event *mod.ModifierFiredEvent) {
	mod.Log("entity", event.GetCaster(), "is about to heal entity", event.GetTarget())
})
```

</Samples>

A cast arrives once it takes effect, after any cast time. Casts arrive for a
hero's abilities and for the items a player activates. The sample checks `slot`
because only a player's hero casts from a slot, and only such a cast carries a
name. The game fires `PRE_UNIT_HEALED` before every heal and every tick of
regeneration. So a real handler for it should do less work than the sample,
which writes a log line for every one.

| Cast field | Holds                                                                                     |
| ---------- | ----------------------------------------------------------------------------------------- |
| `caster`   | The entity that cast it, a hero for a player's cast.                                      |
| `player`   | The player whose hero cast it, or slot -1.                                                |
| `ability`  | The ability's entity, as the hero's `abilities()` reports it.                             |
| `name`     | The ability's name, such as `citadel_ability_hornet_snipe`, when a player's hero cast it. |
| `slot`     | The ability's place on the hero, or -1 when the caster is not a player's hero.            |
| `target`   | The entity the cast was aimed at, or 0.                                                   |

`onModifierEvent` accepts any `EModifierEvent` name, with or without the
`MODIFIER_EVENT_` prefix. The event's `name` matches the string you registered
with. If the running game does not define that event name, the host logs a
warning once and the handler never fires. Register modifier event handlers at
mod startup, since the server collects subscriptions when starting the mod.

Every modifier event has the same four fields, but which ones the game fills
depends on the event:

- `caster` is the entity the event is about.
- `player` is the player whose hero the caster is, or slot -1.
- `target` is the other entity, such as the victim of a kill.
- `ability` is the entity that the event gives as its source.

An entity field the game does not fill is 0. The events below have been seen in
a running game. When one of them has a typed call for the same moment, use the
call, which gives each field its own name:

| Event                | Fills                                           | Typed call      |
| -------------------- | ----------------------------------------------- | --------------- |
| `PLAYER_RESPAWNED`   | `caster`, the hero that respawned.              | `onSpawn`       |
| `PLAYER_KILLED`      | `caster`, the killer, and `target`, the victim. | `onKill`        |
| `KILLED`             | `caster`, the unit that died.                   | `onKill`        |
| `HERO_LEVEL_CHANGED` | `caster`, the hero.                             | `onLevelChange` |
| `UNIT_HEALED`        | Fires very often, so keep its handler short.    | `onHeal`        |
| `PRE_UNIT_HEALED`    | `caster`, the healer, and `target`, the healed. | None            |

## Heals, levels and melee

Your mod can hear each heal, each new level, each melee swing and each parry:

<Samples>

```ts
import { log, onHeal, onLevelChange } from 'modlock'

onHeal(({ targetPlayer, applied }) => {
  if (targetPlayer.slot >= 0) {
    log(`slot ${targetPlayer.slot} regained ${applied} health`)
  }
})

onLevelChange(({ player, level }) => {
  log(`slot ${player.slot} reached level ${level}`)
})
```

```luau
modlock.onHeal(function(heal)
	if heal.targetPlayer.slot >= 0 then
		modlock.log(`slot {heal.targetPlayer.slot} regained {heal.applied} health`)
	end
end)

modlock.onLevelChange(function(change)
	modlock.log(`slot {change.player.slot} reached level {change.level}`)
end)
```

```python
@modlock.on_heal
def healed(heal: modlock.HealEvent) -> None:
    if heal.target_player.slot >= 0:
        modlock.log(f"slot {heal.target_player.slot} regained {heal.applied} health")


@modlock.on_level_change
def leveled(change: modlock.LevelEvent) -> None:
    modlock.log(f"slot {change.player.slot} reached level {change.level}")
```

```go
mod.OnHeal(func(heal *mod.HealEvent) {
	if heal.GetTargetPlayer() >= 0 {
		mod.Log("slot", heal.GetTargetPlayer(), "regained", heal.GetApplied(), "health")
	}
})

mod.OnLevelChange(func(change *mod.LevelEvent) {
	mod.Log("slot", change.GetPlayer(), "reached level", change.GetLevel())
})
```

</Samples>

The sample filters to player heroes (`slot >= 0`), since `-1` indicates a
non-player unit. `onHeal` reports burst heals from abilities, items, or
lifesteal; passive regeneration (including Siphon Life), mod-applied `heal()`
calls, and heals on full-health targets do not fire it. To observe
regeneration ticks, subscribe to the `PRE_UNIT_HEALED`
[modifier event](#casts-and-modifier-events), which fires before every heal and
regeneration tick without an amount. All four events are delivered before the
next frame. Python uses `snake_case` field names, and Go exposes each player as
a slot (`heal.GetTargetPlayer()`).

| Field            | On           | Holds                                                                             |
| ---------------- | ------------ | --------------------------------------------------------------------------------- |
| `target`         | heal         | The entity healed.                                                                |
| `targetPlayer`   | heal         | The player whose hero was healed, or slot -1 for a unit.                          |
| `healer`         | heal         | The entity the heal came from, or 0 when the game gives none.                     |
| `healerPlayer`   | heal         | The player whose hero the healer is, or slot -1.                                  |
| `ability`        | heal         | The ability or item that healed, or 0.                                            |
| `applied`        | heal         | The health the heal restored.                                                     |
| `requested`      | heal         | The heal's whole amount, more than `applied` when the target reached full health. |
| `hero`           | level        | The hero that leveled.                                                            |
| `level`          | level        | The hero's new level.                                                             |
| `player`         | level, parry | The player whose hero leveled or parried, or slot -1.                             |
| `parrier`        | parry        | The hero that parried.                                                            |
| `attacker`       | melee, parry | The hero or unit that swung.                                                      |
| `attackerPlayer` | melee, parry | The player whose hero swung, or slot -1.                                          |
| `heavy`          | melee        | True for a charged swing, false for a light one.                                  |
| `hits`           | melee        | The entities the swing struck, empty for a miss.                                  |

## Pings

Mods intercept each player ping before anyone sees it and can block it. This covers pings from the comms wheel, a quick ping key, the
minimap or an ability. A blocked ping reaches nobody, its sender included, so
no one gets its marker, sound, voice line or chat line. This mod silences the
comms wheel of each player who typed `/quiet`:

<Samples>

```ts
import { command, onPing } from 'modlock'

const quiet = new Set<number>()

command('quiet', (player) => {
  quiet.add(player.slot)
})

onPing(({ player, kind }) => {
  if (kind === 'wheel' && quiet.has(player.slot)) {
    return { block: true }
  }
})
```

```luau
local quiet: { [number]: boolean } = {}

modlock.command("quiet", function(player)
	quiet[player.slot] = true
end)

modlock.onPing(function(ping)
	if ping.kind == "wheel" and quiet[ping.player.slot] then
		return { block = true }
	end
	return nil
end)
```

```python
quiet: set[int] = set()


@modlock.command("quiet")
def silence(player: modlock.Player, args: str) -> None:
    quiet.add(player.slot)


@modlock.on_ping
def ping(event: modlock.PingEvent) -> modlock.PingResult | None:
    if event.kind == "wheel" and event.player.slot in quiet:
        return modlock.PingResult(block=True)
    return None
```

```go
quiet := map[int32]bool{}

mod.Command("quiet", func(p mod.Player, args string) {
	quiet[p.Slot()] = true
})

mod.OnPing(func(ping *mod.PingEvent) *mod.PingResult {
	if ping.GetKind() == mod.PingKindWheel && quiet[ping.GetPlayer()] {
		return &mod.PingResult{Block: true}
	}
	return nil
})
```

</Samples>

A handler can instead mute the ping's voice line for some players, and leave
them its marker and sound. `muteVoice` lists those players by slot. The game
puts the voice line's name in the ping, and each player's game speaks the line.
So a muted player hears nothing of it, while the others hear it as usual. This
mod lets each player mute the pings of a player who spams them:

<Samples>

```ts
import { command, onPing, players } from 'modlock'

// muted maps each player to the players whose ping voices they muted.
const muted = new Map<number, Set<number>>()

command('mutepings', (player, args) => {
  const target = players().find((other) => other.name === args.trim())?.player
  if (target) {
    muted.set(
      target.slot,
      (muted.get(target.slot) ?? new Set()).add(player.slot),
    )
  }
})

onPing(({ player }) => ({
  block: false,
  muteVoice: [...(muted.get(player.slot) ?? [])],
}))
```

```luau
local muted: { [number]: { number } } = {}

modlock.command("mutepings", function(player, args)
	for _, other in modlock.players() do
		if other.name == args then
			local slot = other.player.slot
			muted[slot] = muted[slot] or {}
			table.insert(muted[slot], player.slot)
		end
	end
end)

modlock.onPing(function(ping)
	return { block = false, muteVoice = muted[ping.player.slot] }
end)
```

```python
muted: dict[int, list[int]] = {}


@modlock.command("mutepings")
def mute(player: modlock.Player, args: str) -> None:
    for other in modlock.players():
        if other.name == args.strip():
            muted.setdefault(other.player.slot, []).append(player.slot)


@modlock.on_ping
def ping(event: modlock.PingEvent) -> modlock.PingResult | None:
    return modlock.PingResult(mute_voice=muted.get(event.player.slot, []))
```

```go
muted := map[int32][]int32{}

mod.Command("mutepings", func(p mod.Player, args string) {
	roster, _ := mod.Players()
	for _, other := range roster {
		if other.GetName() == strings.TrimSpace(args) {
			muted[other.GetPlayer()] = append(muted[other.GetPlayer()], p.Slot())
		}
	}
})

mod.OnPing(func(ping *mod.PingEvent) *mod.PingResult {
	return &mod.PingResult{MuteVoice: muted[ping.GetPlayer()]}
})
```

</Samples>

The handler runs inside the server's handling of the ping, so it must answer
at once. Put other work, such as a message to the player, in a timer. When
several handlers answer, the first one that blocks the ping decides. When none
blocks it, each player that any handler mutes gets the ping without its
voice.

| Field     | Meaning                                                                     |
| --------- | --------------------------------------------------------------------------- |
| `player`  | The player who pinged.                                                      |
| `kind`    | `map`, `wheel`, `quickResponse` or `ability`.                               |
| `option`  | The comms wheel option's id from `scripts/ping_wheel_messages.vdata`, or 0. |
| `concept` | The game's `EPingConcept` value for the ping, which selects its voice line. |
