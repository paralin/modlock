Each step adds a few lines to the project `modlock new` wrote, and you can
try each one as soon as it saves. Together they build a small brawl with a
countdown, a score, a HUD and a sign. The [examples](/start/examples) carry
the same ideas into complete game modes you can play.

## 1. Answer a command

`command` runs a handler when a player types `/name` in chat.

<Samples>

```ts
import { command, toast } from 'modlock'

command('roll', (player, args) => {
  const sides = Number(args) || 6
  const roll = 1 + Math.floor(Math.random() * sides)
  toast(player, `You rolled ${roll} out of ${sides}.`)
})
```

```luau
local modlock = require("@modlock")

modlock.command("roll", function(player, args)
	local sides = math.max(1, math.floor(tonumber(args) or 6))
	local roll = math.random(1, sides)
	player:chat(`You rolled {roll} out of {sides}.`)
end)
```

```python
import random

import modlock


@modlock.command("roll")
def roll(player: modlock.Player, args: str) -> None:
    sides = int(args) if args.isdigit() and int(args) > 0 else 6
    player.chat(f"You rolled {random.randint(1, sides)} out of {sides}.")
```

```go
package main

import (
	"math/rand/v2"
	"strconv"

	"github.com/paralin/modlock/mod"
)

func init() {
	mod.Command("roll", func(p mod.Player, args string) {
		sides, err := strconv.Atoi(args)
		if err != nil || sides < 1 {
			sides = 6
		}
		roll := 1 + rand.IntN(sides)
		_ = p.Chat("You rolled " + strconv.Itoa(roll) + " out of " + strconv.Itoa(sides) + ".")
	})
}

func main() {}
```

</Samples>

Type `/roll 20`.

<Only lang="ts">
  `toast` shows a notice that fades. It is part of the TypeScript library's
  [interfaces](/guides/interfaces).
</Only>

## 2. Read the hero

A player's pawn is their hero as it is this frame: its health, team,
position, aim direction and souls. There is none until the hero spawns, and
a hero whose health has reached zero is dead.

<Samples>

```ts
import { command, toast } from 'modlock'

command('where', (player) => {
  const pawn = player.pawn()
  if (pawn === undefined || pawn.health <= 0) {
    toast(player, 'You have no living hero.')
    return
  }
  const { x, y } = pawn.position
  toast(
    player,
    `You stand at ${x.toFixed(0)}, ${y.toFixed(0)} with ${pawn.health} health.`,
  )
})
```

```luau
modlock.command("where", function(player)
	local pawn = player:pawn()
	if pawn == nil or pawn.health <= 0 then
		player:chat("You have no living hero.")
		return
	end
	local x, y = math.round(pawn.position.x), math.round(pawn.position.y)
	player:chat(`You stand at {x}, {y} with {pawn.health} health.`)
end)
```

```python
@modlock.command("where")
def where(player: modlock.Player, args: str) -> None:
    pawn = player.pawn()
    if pawn is None or pawn.health <= 0:
        player.chat("You have no living hero.")
        return
    x, y = pawn.position.x, pawn.position.y
    player.chat(f"You stand at {x:.0f}, {y:.0f} with {pawn.health} health.")
```

```go
mod.Command("where", func(p mod.Player, args string) {
	pawn, err := p.Pawn()
	if err != nil || pawn == nil || pawn.GetHealth() <= 0 {
		_ = p.Chat("You have no living hero.")
		return
	}
	at := pawn.GetPosition()
	_ = p.Chat(fmt.Sprintf("You stand at %.0f, %.0f with %d health.",
		at.GetX(), at.GetY(), pawn.GetHealth()))
})
```

</Samples>

## 3. Count down with a timer

A timer runs a function once, some seconds of game time from now. The game
clock stops when the game pauses, so a paused round holds its timers. The
player list holds everyone connected, bots included.

<Samples>

```ts
import { after, command, players } from 'modlock'

command('start', () => {
  for (const { player } of players()) {
    player.freeze(true)
  }
  after(5, () => {
    for (const { player } of players()) {
      player.freeze(false)
      player.announce('Fight!', 'The round is live.')
    }
  })
})
```

```luau
modlock.command("start", function()
	for _, connection in modlock.players() do
		connection.player:freeze(true)
	end
	modlock.after(5, function()
		for _, connection in modlock.players() do
			connection.player:freeze(false)
			connection.player:announce("Fight!", "The round is live.")
		end
	end)
end)
```

```python
@modlock.command("start")
def start(player: modlock.Player, args: str) -> None:
    for connection in modlock.players():
        connection.player.freeze(True)

    def go() -> None:
        for connection in modlock.players():
            connection.player.freeze(False)
            connection.player.announce("Fight!", "The round is live.")

    modlock.after(5, go)
```

```go
mod.Command("start", func(mod.Player, string) {
	connections, _ := mod.Players()
	for _, c := range connections {
		_ = mod.Player{Slot: c.GetPlayer()}.Freeze(nil)
	}
	mod.After(5*time.Second, func() {
		released := false
		connections, _ := mod.Players()
		for _, c := range connections {
			p := mod.Player{Slot: c.GetPlayer()}
			_ = p.Freeze(&released)
			_ = p.Announce("Fight!", "The round is live.")
		}
	})
})
```

</Samples>

`/start` freezes every hero for five seconds, then announces the round.
[Timers](/guides/timers) covers repeating timers and sequences.

## 4. Decide the fights

The damage event sees each hit before it lands: block it, or change its
amount. The damaged event reports each hit after it lands, with the
victim's health before and the health it lost, so a lethal hit is one that
took all of it. Hits refer to heroes by their entity, the same number as
the pawn's `entity`.

<Samples>

```ts
import { onDamage, onDamaged, type Player, players } from 'modlock'

let live = false
const defeats = new Map<number, number>()

// Nobody gets hurt until the round is live.
onDamage(() => (live ? undefined : { block: true }))

// A lethal hit scores for the player who landed it.
onDamaged((hit) => {
  const scorer = playerOf(hit.attacker)
  if (hit.healthLost < hit.healthBefore || scorer === undefined) {
    return
  }
  defeats.set(scorer.slot, (defeats.get(scorer.slot) ?? 0) + 1)
})

function playerOf(entity: number): Player | undefined {
  return players().find(({ player }) => player.pawn()?.entity === entity)
    ?.player
}
```

```luau
local live = false
local defeats: { [number]: number } = {}

local function playerOf(entity: number): modlock.Player?
	for _, connection in modlock.players() do
		local pawn = connection.player:pawn()
		if pawn and pawn.entity == entity then
			return connection.player
		end
	end
	return nil
end

-- Nobody gets hurt until the round is live.
modlock.onDamage(function(hit)
	if live then
		return nil
	end
	return { block = true }
end)

-- A lethal hit scores for the player who landed it.
modlock.onDamaged(function(hit)
	local scorer = playerOf(hit.attacker)
	if hit.healthLost < hit.healthBefore or scorer == nil then
		return
	end
	defeats[scorer.slot] = (defeats[scorer.slot] or 0) + 1
end)
```

```python
live = False
defeats: dict[int, int] = {}


def player_of(entity: int) -> modlock.Player | None:
    for connection in modlock.players():
        pawn = connection.player.pawn()
        if pawn is not None and pawn.entity == entity:
            return connection.player
    return None


# Nobody gets hurt until the round is live.
@modlock.on_damage
def guard(hit: modlock.DamageEvent) -> modlock.DamageResult | None:
    return None if live else modlock.DamageResult(block=True)


# A lethal hit scores for the player who landed it.
@modlock.on_damaged
def scored(hit: modlock.DamagedEvent) -> None:
    scorer = player_of(hit.attacker)
    if hit.health_lost < hit.health_before or scorer is None:
        return
    defeats[scorer.slot] = defeats.get(scorer.slot, 0) + 1
```

```go
var (
	live    bool
	defeats = map[int32]int{}
)

func init() {
	// Nobody gets hurt until the round is live.
	mod.OnDamage(func(hit *mod.DamageEvent) *mod.DamageResult {
		if live {
			return nil
		}
		block := true
		return &mod.DamageResult{Block: &block}
	})

	// A lethal hit scores for the player who landed it.
	mod.OnDamaged(func(hit *mod.DamagedEvent) {
		scorer, ok := playerOf(hit.GetAttacker())
		if hit.GetHealthLost() < hit.GetHealthBefore() || !ok {
			return
		}
		defeats[scorer.Slot]++
	})
}

func playerOf(entity uint32) (mod.Player, bool) {
	connections, _ := mod.Players()
	for _, c := range connections {
		p := mod.Player{Slot: c.GetPlayer()}
		if pawn, err := p.Pawn(); err == nil && pawn != nil && pawn.GetEntity() == entity {
			return p, true
		}
	}
	return mod.Player{}, false
}
```

</Samples>

Set `live` in step 3's timer, where the round goes live. This map keys
scores by slot. When players come and go, key them by slot and
`generation`: the generation changes when someone new takes the slot.

Each connection also carries `roles`, which say who builds and tests the mod:
`developer` and `tester`. `modlock dev` and `modlock play` give every local
player both, and a platform that hosts the mod sends its own over the control
link, one table per mod. A role decides only what the mod shows, such as a menu still in testing;
whatever guards saved data must check the player where the data lives.

## 5. Show a HUD

<Callout>
  Interfaces are written in TypeScript today. A mod in another language shows
  text with `chat`, `centerText` and `announce`, and world text from the next
  step.
</Callout>

An interface is JSX in a `.tsx` module. Put this HUD in `hud.tsx` beside
`main.ts`:

```tsx
import { type Player, show } from 'modlock'

export function drawScore(player: Player, defeats: number): void {
  show(player, <Score defeats={defeats} />)
}

function Score(props: { defeats: number }) {
  return (
    <panel
      style={{
        horizontalAlign: 'right',
        verticalAlign: 'top',
        margin: [80, 24],
        padding: 10,
        background: '#101820d0',
        borderRadius: 6,
      }}
    >
      <label style={{ fontSize: 24, bold: true }}>
        {props.defeats} defeats
      </label>
    </panel>
  )
}
```

and draw it from `main.ts` on every frame:

```ts
import { drawScore } from './hud'

onFrame(() => {
  for (const { player } of players()) {
    drawScore(player, defeats.get(player.slot) ?? 0)
  }
})
```

`show` sends only what changed since its last call, so redraw every player's HUD
from your state each frame. A `<button onPress={(player) => ready(player)}>`
calls back into the mod when a player presses it.

[Interfaces](/guides/interfaces) lists every element and style.

## 6. Place things in the world

A mod places text, props and effects, each an object to move, change or
remove. A new map removes everything, so forget your objects in the world
event, which runs as each world loads.

<Samples>

```ts
import { command, createText, onWorld, type WorldObject } from 'modlock'

let sign: WorldObject | undefined

// A new world has removed the sign.
onWorld(() => {
  sign = undefined
})

command('sign', (player, args) => {
  const pawn = player.pawn()
  if (pawn === undefined) {
    return
  }
  const { x, y, z } = pawn.position
  sign?.remove()
  sign = createText({
    text: args || 'Hello!',
    position: { x, y, z: z + 120 },
    faceCamera: true,
  })
})
```

```luau
local sign: modlock.WorldObject? = nil

-- A new world has removed the sign.
modlock.onWorld(function()
	sign = nil
end)

modlock.command("sign", function(player, args)
	local pawn = player:pawn()
	if pawn == nil then
		return
	end
	if sign then
		sign:remove()
	end
	local at = pawn.position
	sign = modlock.createText({
		text = if args ~= "" then args else "Hello!",
		position = { x = at.x, y = at.y, z = at.z + 120 },
		faceCamera = true,
	})
end)
```

```python
sign: modlock.WorldObject | None = None


# A new world has removed the sign.
@modlock.on_world
def forget(map_name: str) -> None:
    global sign
    sign = None


@modlock.command("sign")
def place(player: modlock.Player, args: str) -> None:
    global sign
    pawn = player.pawn()
    if pawn is None:
        return
    if sign is not None:
        sign.remove()
    at = pawn.position
    sign = modlock.create_text(
        modlock.TextOptions(
            text=args or "Hello!",
            position=modlock.Vector(at.x, at.y, at.z + 120),
            face_camera=True,
        )
    )
```

```go
var sign *mod.WorldObject

func init() {
	// A new world has removed the sign.
	mod.OnWorld(func(string) { sign = nil })

	mod.Command("sign", func(p mod.Player, args string) {
		pawn, err := p.Pawn()
		if err != nil || pawn == nil {
			return
		}
		if sign != nil {
			_ = sign.Remove()
		}
		if args == "" {
			args = "Hello!"
		}
		at, faceCamera := pawn.GetPosition(), true
		text, err := mod.CreateText(&mod.TextOptions{
			Text:       args,
			Position:   &mod.Vector{X: at.GetX(), Y: at.GetY(), Z: at.GetZ() + 120},
			FaceCamera: &faceCamera,
		})
		if err == nil {
			sign = &text
		}
	})
}
```

</Samples>

Precache the heroes and models your mode uses when the mod starts, so they
load with the world. [The world](/guides/world) covers models, effects and
fog.

## 7. Let players choose, and keep score

A setting is a choice each player keeps between matches, such as where the score
sits. A metric is a total the mod keeps for each player, such as their defeats.
Declare both in `mod.json` under `settings` and `metrics`:

```json
"settings": [
  {
    "key": "corner",
    "label": "Score corner",
    "kind": "KIND_CHOICE",
    "choices": [
      { "value": "right", "label": "Top right" },
      { "value": "left", "label": "Top left" }
    ]
  }
],
"metrics": [{ "name": "brawl.defeats", "kind": "KIND_COUNT" }]
```

Then read the player's setting when you draw the HUD, and add to the metric
when they score:

<Samples>

```ts
const corner = player.setting('corner') === 'left' ? 'left' : 'right'
scorer.addMetric('brawl.defeats', 1)
```

```luau
local corner = if player:setting("corner") == "left" then "left" else "right"
scorer:addMetric("brawl.defeats", 1)
```

```python
corner = "left" if player.setting("corner") == "left" else "right"
scorer.add_metric("brawl.defeats", 1)
```

```go
corner, _ := p.Setting("corner")
_ = scorer.AddMetric("brawl.defeats", 1, nil)
```

</Samples>

A board ranks players against each other, such as the fastest clear of each
course. [Settings, metrics and boards](/guides/data) shows how to declare
one and report to it.

## 8. Split it into modules

As the mode grows, keep `main.ts` a short list of wiring and the game's state in
one object the handlers call into. Split the rest by concern. `modlock build`
bundles and type-checks every module `main.ts` imports.

```ts
import { command, onDamage, onFrame, onStart, onWorld } from 'modlock'

import { drawScoreboards } from './hud'
import { Match } from './match'

const match = new Match()

onStart(() => match.start())
onWorld(() => match.world())
onDamage((hit) => match.damage(hit))
onFrame((frame) => {
  match.frame(frame.timeSeconds)
  drawScoreboards(match)
})
command('ready', (player) => match.ready(player))
```

[`examples/arena`](https://github.com/paralin/modlock/tree/master/examples/arena) is laid out this way. An event that takes
more than 500 ms of processor time, counting the game calls it makes but not
a stall in which the server waits, such as paging on a machine short of
memory, still finishes. The host logs where its time went, warns the players
in chat, and for the next five minutes lets any of them restart the mod with
`/restartmod`, in case it is stuck. Only a runaway event past five seconds
stops the mod. A mod that stops, by a crash or a runaway event, clears what it
placed and shows and stays stopped: the host warns in chat each minute until
a player types `/restartmod` or a reload replaces it. A restart starts the
same build in a new instance, as a reload does.

## 9. Test a whole round

The sandbox runs commands and interfaces in seconds; use it while you write
the rules. To test a whole round in the game, let the mod test itself. Started
with `modlock play --arg probe`, it seats bots with `addBot`, plays the round,
checks the outcome, logs PASS or FAIL and quits with `serverCommand('quit')`.
[`examples/arena`](https://github.com/paralin/modlock/blob/master/examples/arena/probe.ts) does this.

`ProbeRun` does the bookkeeping. Call `tick` each frame with the step the
probe waits on and a check that returns `true` to pass, a message to fail, or
`false` to wait. It logs each new step once, fails the run after its timeout,
then logs `<name> probe PASS` or `FAIL` and quits:

```ts
const run = new ProbeRun('Duel', 120)
onFrame((frame) => {
  run.tick(frame.timeSeconds, 'two fighters', () => fighters() >= 2)
})
```

[Testing](/guides/testing) has more. When the mode plays well,
[publish it](/reference/publishing).
