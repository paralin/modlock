Each step adds a few lines to the project that `modlock new` made. When you save, `modlock dev` reloads the mod, and you can try the step right away. By the last step you have a small brawl with a countdown, a score, a HUD and a sign. The [examples](/start/examples) build the same ideas into complete game modes you can play.

Put a step's import lines at the top of the main file and the rest of its code at the end. A step imports only the names it adds, so keep the imports from earlier steps.

## 1. Answer a command

`command` registers a handler, and the server calls it whenever a player types `/` and the command's name in chat.

<Samples>

```ts
import { toast } from 'modlock'

command('roll', (player, args) => {
  const sides = Number(args) || 6
  const roll = 1 + Math.floor(Math.random() * sides)
  toast(player, `You rolled ${roll} out of ${sides}.`)
})
```

```luau
local ui = require("@modlock/ui")

modlock.command("roll", function(player, args)
	local sides = math.max(1, math.floor(tonumber(args) or 6))
	local roll = math.random(1, sides)
	ui.toast(player, `You rolled {roll} out of {sides}.`)
end)
```

```python
import random

from modlock.ui import toast


@modlock.command("roll")
def roll(player: modlock.Player, args: str) -> None:
    sides = int(args) if args.isdigit() and int(args) > 0 else 6
    toast(player, f"You rolled {random.randint(1, sides)} out of {sides}.")
```

```go
import (
	"math/rand/v2"
	"strconv"

	"github.com/paralin/modlock/mod/ui"
)

func init() {
	mod.Command("roll", func(p mod.Player, args string) {
		sides, err := strconv.Atoi(args)
		if err != nil || sides < 1 {
			sides = 6
		}
		roll := 1 + rand.IntN(sides)
		ui.Toast(p, "You rolled "+strconv.Itoa(roll)+" out of "+strconv.Itoa(sides)+".")
	})
}
```

</Samples>

Type `/roll 20` in the game's chat to try it.

`toast` shows a notice that fades, one of the [interface](/guides/interfaces) calls.

## 2. Read the hero

A player's pawn is their hero as it is in the current frame: its health, team, position, aim direction and souls. `pawn()` returns nothing until the hero has spawned, and a hero whose health is zero is dead.

<Samples>

```ts
command('where', (player) => {
  const pawn = player.pawn()
  if (pawn === undefined || pawn.health <= 0) {
    toast(player, 'You have no living hero.')
    return
  }
  const { x, y } = pawn.position
  toast(
    player,
    `You stand at ${x.toFixed(0)}, ${y.toFixed(0)} with ${pawn.health} health.`,
  )
})
```

```luau
modlock.command("where", function(player)
	local pawn = player:pawn()
	if pawn == nil or pawn.health <= 0 then
		player:chat("You have no living hero.")
		return
	end
	local x, y = math.round(pawn.position.x), math.round(pawn.position.y)
	player:chat(`You stand at {x}, {y} with {pawn.health} health.`)
end)
```

```python
@modlock.command("where")
def where(player: modlock.Player, args: str) -> None:
    pawn = player.pawn()
    if pawn is None or pawn.health <= 0:
        player.chat("You have no living hero.")
        return
    x, y = pawn.position.x, pawn.position.y
    player.chat(f"You stand at {x:.0f}, {y:.0f} with {pawn.health} health.")
```

```go
import "fmt"

func init() {
	mod.Command("where", func(p mod.Player, args string) {
		pawn, err := p.Pawn()
		if err != nil || pawn == nil || pawn.GetHealth() <= 0 {
			_ = p.Chat("You have no living hero.")
			return
		}
		at := pawn.GetPosition()
		_ = p.Chat(fmt.Sprintf("You stand at %.0f, %.0f with %d health.",
			at.GetX(), at.GetY(), pawn.GetHealth()))
	})
}
```

</Samples>

## 3. Count down with a timer

`after` calls a function once, a number of seconds of game time from now. The game clock stops while the game is paused, so a paused round also holds its timers. `players()` lists everyone connected, bots included, and `live` records that the round has started so that the next step can read it.

<Samples>

```ts
import { after, players } from 'modlock'

let live = false

command('start', () => {
  for (const { player } of players()) {
    player.freeze(true)
  }
  after(5, () => {
    live = true
    for (const { player } of players()) {
      player.freeze(false)
      player.announce('Fight!', 'The round is live.')
    }
  })
})
```

```luau
local live = false

modlock.command("start", function()
	for _, connection in modlock.players() do
		connection.player:freeze(true)
	end
	modlock.after(5, function()
		live = true
		for _, connection in modlock.players() do
			connection.player:freeze(false)
			connection.player:announce("Fight!", "The round is live.")
		end
	end)
end)
```

```python
live = False


@modlock.command("start")
def countdown(player: modlock.Player, args: str) -> None:
    for connection in modlock.players():
        connection.player.freeze(True)

    def go() -> None:
        global live
        live = True
        for connection in modlock.players():
            connection.player.freeze(False)
            connection.player.announce("Fight!", "The round is live.")

    modlock.after(5, go)
```

```go
var live bool

func init() {
	mod.Command("start", func(mod.Player, string) {
		connections, _ := mod.Players()
		for _, c := range connections {
			_ = mod.Player{Slot: c.GetPlayer()}.Freeze(nil)
		}
		mod.After(5*time.Second, func() {
			live = true
			released := false
			connections, _ := mod.Players()
			for _, c := range connections {
				p := mod.Player{Slot: c.GetPlayer()}
				_ = p.Freeze(&released)
				_ = p.Announce("Fight!", "The round is live.")
			}
		})
	})
}
```

</Samples>

Type `/start`: every hero freezes for five seconds, and then the mod announces the round. [Timers](/guides/timers) explains repeating timers and sequences.

## 4. Decide the fights

Your mod can handle two events for each hit. The damage event fires before the game applies the hit, and your handler can block it or change its amount. The damaged event fires afterwards with the victim's health before the hit and the health it lost. A hit that lost all of that health was lethal. In a hit, each hero is an entity number, the same number as the pawn's `entity`, and `playerOf` finds the player by that number.

<Samples>

```ts
import { onDamage, onDamaged, type Player } from 'modlock'

const defeats = new Map<number, number>()

// Nobody gets hurt until the round is live.
onDamage(() => (live ? undefined : { block: true }))

// A lethal hit scores for the player who landed it.
onDamaged((hit) => {
  const scorer = playerOf(hit.attacker)
  if (hit.healthLost < hit.healthBefore || scorer === undefined) {
    return
  }
  defeats.set(scorer.slot, (defeats.get(scorer.slot) ?? 0) + 1)
})

function playerOf(entity: number): Player | undefined {
  return players().find(({ player }) => player.pawn()?.entity === entity)
    ?.player
}
```

```luau
local defeats: { [number]: number } = {}

local function playerOf(entity: number): modlock.Player?
	for _, connection in modlock.players() do
		local pawn = connection.player:pawn()
		if pawn and pawn.entity == entity then
			return connection.player
		end
	end
	return nil
end

-- Nobody gets hurt until the round is live.
modlock.onDamage(function(hit)
	if live then
		return nil
	end
	return { block = true }
end)

-- A lethal hit scores for the player who landed it.
modlock.onDamaged(function(hit)
	local scorer = playerOf(hit.attacker)
	if hit.healthLost < hit.healthBefore or scorer == nil then
		return
	end
	defeats[scorer.slot] = (defeats[scorer.slot] or 0) + 1
end)
```

```python
defeats: dict[int, int] = {}


def player_of(entity: int) -> modlock.Player | None:
    for connection in modlock.players():
        pawn = connection.player.pawn()
        if pawn is not None and pawn.entity == entity:
            return connection.player
    return None


# Nobody gets hurt until the round is live.
@modlock.on_damage
def guard(hit: modlock.DamageEvent) -> modlock.DamageResult | None:
    return None if live else modlock.DamageResult(block=True)


# A lethal hit scores for the player who landed it.
@modlock.on_damaged
def scored(hit: modlock.DamagedEvent) -> None:
    scorer = player_of(hit.attacker)
    if hit.health_lost < hit.health_before or scorer is None:
        return
    defeats[scorer.slot] = defeats.get(scorer.slot, 0) + 1
```

```go
var defeats = map[int32]int{}

func init() {
	// Nobody gets hurt until the round is live.
	mod.OnDamage(func(hit *mod.DamageEvent) *mod.DamageResult {
		if live {
			return nil
		}
		block := true
		return &mod.DamageResult{Block: &block}
	})

	// A lethal hit scores for the player who landed it.
	mod.OnDamaged(func(hit *mod.DamagedEvent) {
		scorer, ok := playerOf(hit.GetAttacker())
		if hit.GetHealthLost() < hit.GetHealthBefore() || !ok {
			return
		}
		defeats[scorer.Slot]++
	})
}

func playerOf(entity uint32) (mod.Player, bool) {
	connections, _ := mod.Players()
	for _, c := range connections {
		p := mod.Player{Slot: c.GetPlayer()}
		if pawn, err := p.Pawn(); err == nil && pawn != nil && pawn.GetEntity() == entity {
			return p, true
		}
	}
	return mod.Player{}, false
}
```

</Samples>

This map keys scores by slot. A slot is reused when one player leaves and another joins. In a mode where players come and go, key scores by slot and `generation`, which changes each time someone new takes the slot.

Each connection also has [`roles`](/guides/players#roles), which tell you which players build and test the mod.

## 5. Show a HUD

An interface is a tree of elements. In TypeScript you write it as JSX in a `.tsx` module, and in the other languages you build it with calls to the `ui` library. Put this HUD in its own module beside the main one: `hud.tsx`, `hud.luau` or `hud.py`, or in Go a `hud.go` in the same package:

<Samples>

```tsx
import { type Player, show } from 'modlock'

export function drawScore(player: Player, defeats: number): void {
  show(player, <Score defeats={defeats} />)
}

function Score(props: { defeats: number }) {
  return (
    <panel
      style={{
        horizontalAlign: 'right',
        verticalAlign: 'top',
        margin: [80, 24],
        padding: 10,
        background: '#101820d0',
        borderRadius: 6,
      }}
    >
      <label style={{ fontSize: 24, bold: true }}>
        {props.defeats} defeats
      </label>
    </panel>
  )
}
```

```luau
local modlock = require("@modlock")
local ui = require("@modlock/ui")

local hud = {}

function hud.drawScore(player: modlock.Player, defeats: number)
	ui.show(player, ui.panel {
		style = {
			horizontalAlign = "right",
			verticalAlign = "top",
			margin = { 80, 24 },
			padding = 10,
			background = "#101820d0",
			borderRadius = 6,
		},
		ui.label { style = { fontSize = 24, bold = true }, `{defeats} defeats` },
	})
end

return hud
```

```python
import modlock
from modlock.ui import label, panel, show


def draw_score(player: modlock.Player, defeats: int) -> None:
    show(player, panel(style={
        "horizontal_align": "right",
        "vertical_align": "top",
        "margin": (80, 24),
        "padding": 10,
        "background": "#101820d0",
        "border_radius": 6,
    })[
        label(style={"font_size": 24, "bold": True})[f"{defeats} defeats"],
    ])
```

```go

	"strconv"

	"github.com/paralin/modlock/mod"
	"github.com/paralin/modlock/mod/ui"
)

		Style: ui.Style{
			HorizontalAlign: ui.Right,

			Background:      "#101820d0",
			BorderRadius:    6,
		},

			Style: ui.Style{FontSize: 24, Bold: true},
			Text:  strconv.Itoa(defeats) + " defeats",
		},
	})
}
```

</Samples>

Draw the HUD from the main module on every frame:

<Samples>

```ts
import { drawScore } from './hud'

onFrame(() => {
  for (const { player } of players()) {
    drawScore(player, defeats.get(player.slot) ?? 0)
  }
})
```

```luau
modlock.onFrame(function()
	for _, connection in modlock.players() do
		hud.drawScore(connection.player, defeats[connection.player.slot] or 0)
	end
end)
```

```python

def draw(frame: modlock.FrameEvent) -> None:
    for connection in modlock.players():
        hud.draw_score(connection.player, defeats.get(connection.player.slot, 0))
```

```go
		connections, _ := mod.Players()

```

</Samples>

Redraw every player's HUD from your state on each frame. `show` compares the tree with the one it sent last and sends the player only what changed, so a redraw on every frame costs little. When a player presses a button, the mod calls its `onPress` with that player.

## 6. Place things in the world

Your mod can place text, props and effects in the world, and each call returns an object you can move, change or remove. Loading a new map removes all of them. Drop your references in the world event, which fires as each world loads. When the mod starts, precache the heroes and models your mode uses, so the game loads them with the world. [The world](/guides/world) explains models, effects and fog.

<Samples>

```ts
import { createText, onWorld, type WorldObject } from 'modlock'

let sign: WorldObject | undefined

// A new world has removed the sign.
onWorld(() => {
  sign = undefined
})

command('sign', (player, args) => {
  const pawn = player.pawn()
  if (pawn === undefined) {
    return
  }
  const { x, y, z } = pawn.position
  sign?.remove()
  sign = createText({
    text: args || 'Hello!',
    position: { x, y, z: z + 120 },
    faceCamera: true,
  })
})
```

```luau
local sign: modlock.WorldObject? = nil

-- A new world has removed the sign.
modlock.onWorld(function()
	sign = nil
end)

modlock.command("sign", function(player, args)
	local pawn = player:pawn()
	if pawn == nil then
		return
	end
	if sign then
		sign:remove()
	end
	local at = pawn.position
	sign = modlock.createText({
		text = if args ~= "" then args else "Hello!",
		position = { x = at.x, y = at.y, z = at.z + 120 },
		faceCamera = true,
	})
end)
```

```python
sign: modlock.WorldObject | None = None


# A new world has removed the sign.
@modlock.on_world
def forget(map_name: str) -> None:
    global sign
    sign = None


@modlock.command("sign")
def place(player: modlock.Player, args: str) -> None:
    global sign
    pawn = player.pawn()
    if pawn is None:
        return
    if sign is not None:
        sign.remove()
    at = pawn.position
    sign = modlock.create_text(
        modlock.TextOptions(
            text=args or "Hello!",
            position=modlock.Vector(at.x, at.y, at.z + 120),
            face_camera=True,
        )
    )
```

```go
var sign *mod.WorldObject

func init() {
	// A new world has removed the sign.
	mod.OnWorld(func(string) { sign = nil })

	mod.Command("sign", func(p mod.Player, args string) {
		pawn, err := p.Pawn()
		if err != nil || pawn == nil {
			return
		}
		if sign != nil {
			_ = sign.Remove()
		}
		if args == "" {
			args = "Hello!"
		}
		at, faceCamera := pawn.GetPosition(), true
		text, err := mod.CreateText(&mod.TextOptions{
			Text:       args,
			Position:   &mod.Vector{X: at.GetX(), Y: at.GetY(), Z: at.GetZ() + 120},
			FaceCamera: &faceCamera,
		})
		if err == nil {
			sign = &text
		}
	})
}
```

</Samples>

## 7. Let players choose, and keep score

A setting is a choice each player makes once and keeps between matches, such as where the score sits on screen. A metric is a running total for each player that the mod adds to, such as their defeats. Declare both in `mod.json`, under `settings` and `metrics`:

```json
"settings": [
  {
    "key": "corner",
    "label": "Score corner",
    "kind": "KIND_CHOICE",
    "choices": [
      { "value": "right", "label": "Top right" },
      { "value": "left", "label": "Top left" }
    ]
  }
],
"metrics": [{ "name": "brawl.defeats", "kind": "KIND_COUNT" }]
```

Read the player's setting in `drawScore` and align the panel with it. Add to the metric in the damaged handler once the hit scores:

<Samples>

```ts
const corner = player.setting('corner') === 'left' ? 'left' : 'right'
scorer.addMetric('brawl.defeats', 1)
```

```luau
local corner = if player:setting("corner") == "left" then "left" else "right"
scorer:addMetric("brawl.defeats", 1)
```

```python
corner = "left" if player.setting("corner") == "left" else "right"
scorer.add_metric("brawl.defeats", 1)
```

```go
var corner ui.Align = ui.Right
if choice, _ := p.Setting("corner"); choice == "left" {
	corner = ui.Left
}
_ = scorer.AddMetric("brawl.defeats", 1, nil)
```

</Samples>

A board ranks players against each other, such as by their fastest clear of each course. [Settings, metrics and boards](/guides/data) shows how to declare a board and report scores to it.

## 8. Split it into modules

As the mode grows, give each concern its own module. Keep `main.ts` a short list that registers the handlers, and keep the game's state in one object that the handlers call into. `modlock build` bundles and type-checks every module that `main.ts` imports.

```ts
import { command, onDamage, onFrame, onStart, onWorld } from 'modlock'

import { drawScoreboards } from './hud'
import { Match } from './match'

const match = new Match()

onStart(() => match.start())
onWorld(() => match.world())
onDamage((hit) => match.damage(hit))
onFrame((frame) => {
  match.frame(frame.timeSeconds)
  drawScoreboards(match)
})
command('ready', (player) => match.ready(player))
```

[`examples/arena`](https://github.com/paralin/modlock/tree/master/examples/arena) is laid out this way.

Keep each handler quick. An event that takes more than 500 ms of processor time shows the players a warning. When one takes more than five seconds, the server stops the mod until a player types `/restartmod` or a reload replaces it. [Slow and stopped mods](/guides/reloads#slow-and-stopped-mods) explains both limits.

## 9. Test a whole round

To test a whole round in the game, your mod can add bots, play the round
against them and log PASS or FAIL, as [Testing](/guides/testing#in-the-game)
shows. When the mode plays well, [publish it](/reference/publishing).
