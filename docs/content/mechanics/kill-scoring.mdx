---
title: Kill scoring and respawn
description: Score each kill, bring the fallen back and end the round at a goal.
---

A deathmatch needs three things: who landed the lethal hit, a score for
them, and the victim back in the fight. The damaged event gives the first.
It names heroes by entity, so match each entity back to its player.

<Samples>

```ts
import { after, onDamaged, onPlayerLeave, players } from 'modlock'

const goal = 10
const kills = new Map<number, number>()

// The connection whose hero is this entity.
function connectionOf(entity: number) {
  return players().find(({ player }) => player.pawn()?.entity === entity)
}

onDamaged((hit) => {
  if (hit.healthLost < hit.healthBefore) {
    return // not lethal
  }
  const victim = connectionOf(hit.victim)
  const scorer = connectionOf(hit.attacker)
  if (victim !== undefined) {
    after(3, () => victim.player.respawn())
  }
  if (scorer === undefined || scorer === victim) {
    return
  }
  const total = (kills.get(scorer.player.slot) ?? 0) + 1
  kills.set(scorer.player.slot, total)
  const won = total >= goal
  for (const { player } of players()) {
    if (won) {
      player.announce(`${scorer.name} wins`, 'A new round starts now.')
    } else {
      player.centerText(`${scorer.name}: ${total} / ${goal}`)
    }
  }
  if (won) {
    kills.clear()
  }
})

onPlayerLeave(({ connection }) => {
  kills.delete(connection.player.slot)
})
```

```python
import modlock

GOAL = 10
kills: dict[int, int] = {}


def connection_of(entity: int) -> modlock.Connection | None:
    """The connection whose hero is this entity."""
    for connection in modlock.players():
        pawn = connection.player.pawn()
        if pawn is not None and pawn.entity == entity:
            return connection
    return None


@modlock.on_damaged
def scored(hit: modlock.DamagedEvent) -> None:
    if hit.health_lost < hit.health_before:
        return  # not lethal
    victim = connection_of(hit.victim)
    scorer = connection_of(hit.attacker)
    if victim is not None:
        modlock.after(3, victim.player.respawn)
    if scorer is None or scorer is victim:
        return
    slot = scorer.player.slot
    total = kills.get(slot, 0) + 1
    kills[slot] = total
    won = total >= GOAL
    for connection in modlock.players():
        if won:
            connection.player.announce(f"{scorer.name} wins", "A new round starts now.")
        else:
            connection.player.center_text(f"{scorer.name}: {total} / {GOAL}")
    if won:
        kills.clear()


@modlock.on_player_leave
def left(event: modlock.PlayerLeftEvent) -> None:
    kills.pop(event.connection.player.slot, None)
```

```go
package main

import (
	"fmt"
	"time"

	"github.com/paralin/modlock/mod"
)

const goal = 10

var kills = map[int32]int{}

// connectionOf returns the connection whose hero is this entity.
func connectionOf(entity uint32) *mod.Connection {
	connections, _ := mod.Players()
	for _, c := range connections {
		pawn, err := mod.Player{Slot: c.GetPlayer()}.Pawn()
		if err == nil && pawn != nil && pawn.GetEntity() == entity {
			return c
		}
	}
	return nil
}

func init() {
	mod.OnDamaged(func(hit *mod.DamagedEvent) {
		if hit.GetHealthLost() < hit.GetHealthBefore() {
			return // not lethal
		}
		victim := connectionOf(hit.GetVictim())
		scorer := connectionOf(hit.GetAttacker())
		if victim != nil {
			fallen := mod.Player{Slot: victim.GetPlayer()}
			mod.After(3*time.Second, func() { _ = fallen.Respawn() })
		}
		if scorer == nil || (victim != nil && scorer.GetPlayer() == victim.GetPlayer()) {
			return
		}
		kills[scorer.GetPlayer()]++
		total := kills[scorer.GetPlayer()]
		won := total >= goal
		connections, _ := mod.Players()
		for _, c := range connections {
			p := mod.Player{Slot: c.GetPlayer()}
			if won {
				_ = p.Announce(scorer.GetName()+" wins", "A new round starts now.")
			} else {
				_ = p.CenterText(fmt.Sprintf("%s: %d / %d", scorer.GetName(), total, goal))
			}
		}
		if won {
			clear(kills)
		}
	})

	mod.OnPlayerLeave(func(left *mod.PlayerLeftEvent) {
		delete(kills, left.GetConnection().GetPlayer())
	})
}

func main() {}
```

## Make it yours

- Scores here are keyed by slot and dropped when the player leaves. To keep
  a score through a reconnect, key it by Steam ID instead.
- Pay the scorer with `adjustSouls`, or heal them, to reward a streak.
- Count kills across sessions with a [metric](/guides/data#metrics), or
  rank players with a [board](/guides/data#boards).
- The [`bounty`](https://github.com/paralin/modlock/tree/master/examples/bounty) example builds on this: the leader
  carries a marker, and whoever defeats them takes it.
